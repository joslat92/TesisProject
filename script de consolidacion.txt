$OutputFile = "Auditoria_Proyecto_Tesis.txt"
$ProjectRoot = Get-Location

# Crear el archivo con el encabezado inicial
"======================================================================`n" +
"REPORTE DE AUDITORÍA DE CÓDIGO Y RESULTADOS - TESIS MMACC`n" +
"======================================================================`n" +
"Fecha: $(Get-Date -Format 'yyyy-MM-dd HH:mm')`n" +
"Autor: Jose Luis Agudelo`n" | Out-File -FilePath $OutputFile -Encoding utf8

# Obtener solo los archivos relevantes (Scripts, Configuración y Resultados)
$FilesToBundle = Get-ChildItem -Recurse -Include *.py, *.yaml, *.csv | 
    Where-Object { $_.FullName -notmatch "venv|__pycache__|data\\raw" }

foreach ($File in $FilesToBundle) {
    # Mostrar progreso en consola para que sepas qué está haciendo
    Write-Host "Agregando: $($File.Name)" -ForegroundColor Cyan
    
    # Agregar separador y nombre del archivo
    $RelativePath = $File.FullName.Replace($ProjectRoot.Path, "")
    "`n`n" + ("-" * 70) + "`n" +
    "ARCHIVO: $RelativePath`n" +
    ("-" * 70) + "`n" | Add-Content -Path $OutputFile
    
    # Leer el contenido del script y agregarlo al archivo final
    Get-Content $File.FullName | Add-Content -Path $OutputFile
}

Write-Host "`nConsolidación finalizada con éxito. El archivo es: $OutputFile" -ForegroundColor Green
